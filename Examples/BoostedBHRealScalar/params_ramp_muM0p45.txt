# Initial Data parameters

# Scalar field Initial data
# Units: the BH mass, M = 1 (so scalar_mass = mu M, scalar_lambda = lambda M^2)
# V = 1/2 mu^2 phi^2 + 1/4 lambda phi^4. On the fixed background only
# x = lambda phi0^2 / mu^2 = 0.101 and mu matter (lambda and rho trade off:
# this is rho0 M^2 = 1e-4 with lambda M^2 = 21.8, or 1e-5 with 218)
# c_NR = sqrt(3x/8) = 0.195, c_s0 = c_NR (1 - 1.4 c_NR^2) = 0.184
# r_sg = 2M/c_s0^2 = 59 M, sound-crossing time r_sg/c_s0 = 319 M,
# omega0 = mu sqrt(1 + 3x/4) = 0.467, healing length xi = 1/(mu c_s0) = 12 M
# Shock regime at the end of the ramp (v = 0.2767): beta = mu M/v = 1.63,
# r_BHL/xi = (2M/v^2)/xi = 2.17, xi/r_sg = 0.21 (fluid-like, but transitional;
# mu M = 1 had beta = 3.25, mu M = 0.1 had 0.23)
scalar_mass = 0.45
scalar_amplitude = 0.03065773
scalar_lambda = 21.8

# Boosted BH initial data, boost in x direction
# Initial velocity: one BH of the equal-mass d0 ~ 12M binary of
# Aurrekoetxea et al. 2409.01937 (GRChombo BinaryBH params: |P| = 0.0842 per
# BH of mass 0.5, so v = P/(gamma m) = 0.166)
# The BH velocity is ramped smoothly (C^2 step) from bh_velocity to
# bh_velocity_final over [ramp_start, ramp_start + ramp_duration]:
# Mach number M0 = v/c_s0 goes from 0.90 (subsonic) to 1.50 (supersonic, above
# the detached/attached threshold M_t = (3 f* c_s0)^(-1/3) = 1.40); Mach angle
# asin(c_s0/v) = 42 deg at the end. Ramp starts after ~4 and lasts ~10
# sound-crossing times; ~8 more after the ramp
bh_mass = 1.0
bh_velocity = 0.166 # in unit of c (i.e. 0<v<1)
bh_velocity_final = 0.2767
ramp_start = 1300.0
ramp_duration = 3200.0
